function doPost(e) { try { var folderId = "1VV8BQzppPxrOzFrZvVEVUxMH6rib5I6J"; var folder = DriveApp.getFolderById(folderId); var rawData = e.postData.contents; var data = JSON.parse(rawData); // 1. บันทึกไฟล์ภาพถ่าย (PNG) if (data.image) { var base64Img = data.image.split(",")[1]; var decodedImg = Utilities.base64Decode(base64Img); var blobImg = Utilities.newBlob(decodedImg, "image/png", data.filename || "TSK_photo.png"); var fileImg = folder.createFile(blobImg); fileImg.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } // 2. บันทึกไฟล์ GIF if (data.gif) { var base64Gif = data.gif.split(",")[1]; var decodedGif = Utilities.base64Decode(base64Gif); var blobGif = Utilities.newBlob(decodedGif, "image/gif", data.gifFilename || "TSK_anim.gif"); var fileGif = folder.createFile(blobGif); fileGif.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } return ContentService.createTextOutput("OK").setMimeType(ContentService.MimeType.TEXT); } catch (error) { return ContentService.createTextOutput("ERROR: " + error.toString()).setMimeType(ContentService.MimeType.TEXT); } }
